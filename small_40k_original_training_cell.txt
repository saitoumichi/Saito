def curriculum_shift(epoch):
    return min(1 + (epoch - 1) // 1_000, 40)

class ExperimentHaar3D(nn.Module):
    def __init__(self):
        super().__init__()

        low = torch.tensor(
            [1.0, 1.0], dtype=torch.float32
        ) / np.sqrt(2.0)

        high = torch.tensor(
            [1.0, -1.0], dtype=torch.float32
        ) / np.sqrt(2.0)

        filters = []
        for z_filter in (low, high):
            for y_filter in (low, high):
                for x_filter in (low, high):
                    kernel = (
                        z_filter[:, None, None]
                        * y_filter[None, :, None]
                        * x_filter[None, None, :]
                    )
                    filters.append(kernel)

        self.register_buffer(
            "weight",
            torch.stack(filters).unsqueeze(1),
        )

    def analysis(self, image):
        return F.conv3d(
            image, self.weight,
            stride=2, padding=0,
        )

    def synthesis(self, bands):
        return F.conv_transpose3d(
            bands, self.weight,
            stride=2, padding=0,
        )


experiment_haar = ExperimentHaar3D().to(DEVICE)

experiment_transformer = vxm.layers.SpatialTransformer(
    FULL_SIZE
).to(DEVICE)


def wavelet_input_local(image):
    return experiment_haar.analysis(image)


def full_dvf_to_wavelet_local(flow_full):
    batch_size = flow_full.shape[0]
    flow_images = flow_full.reshape(
        batch_size * 3, 1, *FULL_SIZE
    )
    bands = experiment_haar.analysis(flow_images)
    return bands.reshape(
        batch_size, 24, *LOW_SIZE
    )


def wavelet_to_full_dvf_local(flow_coeff):
    batch_size = flow_coeff.shape[0]
    bands = flow_coeff.reshape(
        batch_size * 3, 8, *LOW_SIZE
    )
    flow_images = experiment_haar.synthesis(bands)
    return flow_images.reshape(
        batch_size, 3, *FULL_SIZE
    )

NB_FEATURES = [
    [32, 64, 64, 64, 64],
    [64, 64, 64, 64, 64, 32, 16, 16],
]


def create_experiment_model():
    net = vxm.networks.VxmDense_128_256_256(
        FULL_SIZE,
        NB_FEATURES,
        int_steps=0,
    ).to(DEVICE)

    old_flow = net.flow

    net.flow = nn.Conv3d(
        in_channels=old_flow.in_channels,
        out_channels=24,
        kernel_size=old_flow.kernel_size,
        stride=old_flow.stride,
        padding=old_flow.padding,
        dilation=old_flow.dilation,
        groups=old_flow.groups,
        bias=True,
    ).to(DEVICE)

    nn.init.normal_(
        net.flow.weight,
        mean=0.0,
        std=1e-5,
    )
    nn.init.zeros_(net.flow.bias)

    return net

# Aと同じ人工教師DVF
# ============================================================

def gaussian_smooth_3d(tensor, sigma=2.0):
    array = tensor.detach().cpu().numpy()

    smoothed = gaussian_filter(
        array,
        sigma=[0, 0, sigma, sigma, sigma],
    )

    return torch.from_numpy(
        np.ascontiguousarray(smoothed)
    ).to(
        device=tensor.device,
        dtype=tensor.dtype,
    )


def generate_artificial_dvf(batch_size, shift):
    coarse_flow = (
        torch.rand(
            batch_size, 3, *COARSE_SIZE,
            device=DEVICE,
            dtype=torch.float32,
        ) * 2.0 - 1.0
    ) * float(shift)

    coarse_flow = gaussian_smooth_3d(
        coarse_flow, sigma=2.0
    )

    global_flow = F.interpolate(
        coarse_flow,
        size=FULL_SIZE,
        mode="trilinear",
        align_corners=False,
    )

    local_d, local_h, local_w = LOCAL_DVF_SIZE

    z_grid = torch.arange(
        local_d, device=DEVICE,
        dtype=torch.float32,
    ).view(local_d, 1, 1)

    y_grid = torch.arange(
        local_h, device=DEVICE,
        dtype=torch.float32,
    ).view(1, local_h, 1)

    x_grid = torch.arange(
        local_w, device=DEVICE,
        dtype=torch.float32,
    ).view(1, 1, local_w)

    local_flow_low = torch.zeros(
        batch_size, 3,
        local_d, local_h, local_w,
        device=DEVICE,
        dtype=torch.float32,
    )

    local_amplitude = min(
        3.0,
        0.1 * float(shift),
    )

    for batch_index in range(batch_size):
        for _ in range(2):
            center_z = torch.empty(
                1, device=DEVICE
            ).uniform_(0, local_d - 1)

            center_y = torch.empty(
                1, device=DEVICE
            ).uniform_(0, local_h - 1)

            center_x = torch.empty(
                1, device=DEVICE
            ).uniform_(0, local_w - 1)

            blob = torch.exp(
                -0.5 * (
                    ((z_grid - center_z) / 1.5).square()
                    + ((y_grid - center_y) / 2.5).square()
                    + ((x_grid - center_x) / 2.5).square()
                )
            )

            direction = torch.randn(
                3, device=DEVICE
            )
            direction = direction / (
                direction.norm() + 1e-8
            )

            amplitude = torch.empty(
                1, device=DEVICE
            ).uniform_(
                0.5 * local_amplitude,
                local_amplitude,
            )

            local_flow_low[batch_index] += (
                direction[:, None, None, None]
                * amplitude
                * blob[None]
            )

    local_flow_full = F.interpolate(
        local_flow_low,
        size=FULL_SIZE,
        mode="trilinear",
        align_corners=False,
    )

    return global_flow + local_flow_full

def coefficient_loss(predicted_coeff, target_coeff):
    difference = (
        predicted_coeff - target_coeff
    ).reshape(
        predicted_coeff.shape[0],
        3, 8, *LOW_SIZE,
    )

    band_mse = difference.square().mean(
        dim=(0, 1, 3, 4, 5)
    )

    return band_mse[0] + band_mse[1:].mean()


def determinant_penalty_patch(flow_patch, margin=0.05):
    """
    flow_patch: (B,3,D,H,W), [z,y,x]
    中心差分、内部voxelだけを評価。
    """
    def diff(t, dim):
        if dim == 2:
            return (
                t[:, :, 2:, 1:-1, 1:-1]
                - t[:, :, :-2, 1:-1, 1:-1]
            ) * 0.5
        if dim == 3:
            return (
                t[:, :, 1:-1, 2:, 1:-1]
                - t[:, :, 1:-1, :-2, 1:-1]
            ) * 0.5
        return (
            t[:, :, 1:-1, 1:-1, 2:]
            - t[:, :, 1:-1, 1:-1, :-2]
        ) * 0.5

    gz = diff(flow_patch, 2)
    gy = diff(flow_patch, 3)
    gx = diff(flow_patch, 4)

    a = 1.0 + gz[:, 0]
    b = gy[:, 0]
    c = gx[:, 0]
    d = gz[:, 1]
    e = 1.0 + gy[:, 1]
    f = gx[:, 1]
    g = gz[:, 2]
    h = gy[:, 2]
    i = 1.0 + gx[:, 2]

    det = (
        a * (e * i - f * h)
        - b * (d * i - f * g)
        + c * (d * h - e * g)
    )

    return F.relu(margin - det).square().mean()


def sampled_jacobian_loss(full_flow):
    """
    フル解像度DVFから3パッチ：
    z先頭、z末尾、ランダム中央。
    毎回パッチ位置を変える。
    """
    _, _, depth, height, width = full_flow.shape
    patch_d, patch_h, patch_w = 16, 48, 48

    if (depth, height, width) != FULL_SIZE:
        raise ValueError(full_flow.shape)

    def random_start(limit):
        return random.randint(0, limit)

    losses = []

    starts = [
        # z上端
        (
            0,
            random_start(height - patch_h),
            random_start(width - patch_w),
        ),
        # z下端
        (
            depth - patch_d,
            random_start(height - patch_h),
            random_start(width - patch_w),
        ),
        # 中央部
        (
            random.randint(16, depth - patch_d - 16),
            random_start(height - patch_h),
            random_start(width - patch_w),
        ),
    ]

    for z0, y0, x0 in starts:
        patch = full_flow[
            :,
            :,
            z0:z0 + patch_d,
            y0:y0 + patch_h,
            x0:x0 + patch_w,
        ]
        losses.append(
            determinant_penalty_patch(
                patch,
                margin=JACOBIAN_MARGIN,
            )
        )

    return torch.stack(losses).mean()

def checkpoint_epoch(path):
    match = re.fullmatch(
        r"checkpoint_epoch_(\d+)\.pth",
        path.name,
    )
    return int(match.group(1)) if match else -1


def cpu_byte_tensor(value):
    if isinstance(value, torch.Tensor):
        return value.detach().cpu().to(torch.uint8)
    return torch.as_tensor(value, dtype=torch.uint8)


def save_checkpoint(epoch, model, optimizer):
    path = CHECKPOINT_DIR / f"checkpoint_epoch_{epoch:05d}.pth"

    if path.exists():
        raise FileExistsError(f"既存checkpointを保護: {path}")

    checkpoint = {
        "epoch": epoch,
        "experiment_id": EXPERIMENT_ID,
        "condition": EXPERIMENT,
        "model_state_dict": model.state_dict(),
        "optimizer_state_dict": optimizer.state_dict(),
        "torch_rng_state": torch.get_rng_state(),
        "numpy_rng_state": np.random.get_state(),
        "python_rng_state": random.getstate(),
        "cuda_rng_state_all": (
            torch.cuda.get_rng_state_all()
            if torch.cuda.is_available() else None
        ),
        "loss_weights": {
            "image": IMAGE_WEIGHT,
            "coefficient": DVF_WEIGHT,
            "jacobian": (
                JACOBIAN_WEIGHT
                if EXPERIMENT == "jacobian" else 0.0
            ),
        },
        "small_motion_probabilities": {
            "zero": ZERO_MOTION_PROB,
            "small": SMALL_MOTION_PROB,
        },
        "jacobian_margin": JACOBIAN_MARGIN,
    }

    torch.save(checkpoint, path)
    return path

def checked_model_forward(moving_bands, fixed_bands):
    # 逆伝播時に中間活性を再計算してVRAMを抑える
    output = activation_checkpoint(
        model,
        moving_bands,
        fixed_bands,
        use_reentrant=False,
        preserve_rng_state=True,
    )
    if isinstance(output, (tuple, list)):
        raise TypeError("モデルが複数値を返しました")
    if tuple(output.shape) != (
        BATCH_SIZE, 24, *LOW_SIZE
    ):
        raise ValueError(
            f"24ch出力形状が想定外: {tuple(output.shape)}"
        )
    return output


# ============================================================
# 学習
# ============================================================

if start_epoch > TOTAL_EPOCHS:
    print("40k checkpointが既にあるため学習しません")
else:
    started_at = time.perf_counter()

    for epoch in tqdm(
        range(start_epoch, TOTAL_EPOCHS + 1),
        desc=EXPERIMENT_ID,
    ):
        model.train()
        shift = curriculum_shift(epoch)

        indices = np.random.choice(
            len(x_train),
            size=BATCH_SIZE,
            replace=False,
        )
        moving = torch.from_numpy(
            np.ascontiguousarray(x_train[indices])
        ).unsqueeze(1).to(
            DEVICE,
            dtype=torch.float32,
        )

        with torch.no_grad():
            teacher_dvf = generate_artificial_dvf(
                BATCH_SIZE, shift
            )

            motion_kind = "original"

            if EXPERIMENT == "small_motion":
                r = torch.rand(
                    (), device=DEVICE
                ).item()

                if r < ZERO_MOTION_PROB:
                    teacher_dvf.zero_()
                    motion_kind = "zero"
                elif r < (
                    ZERO_MOTION_PROB + SMALL_MOTION_PROB
                ):
                    scale = torch.empty(
                        (), device=DEVICE
                    ).uniform_(
                        SMALL_SCALE_MIN,
                        SMALL_SCALE_MAX,
                    )
                    teacher_dvf.mul_(scale)
                    motion_kind = "small"

            # 変更後の教師DVFから両方を作る
            fixed = experiment_transformer(
                moving, teacher_dvf
            )
            target_coeff = full_dvf_to_wavelet_local(
                teacher_dvf
            )

        optimizer.zero_grad(set_to_none=True)

        moving_bands = wavelet_input_local(moving)
        fixed_bands = wavelet_input_local(fixed)

        predicted_coeff = checked_model_forward(
            moving_bands,
            fixed_bands,
        )

        predicted_flow = wavelet_to_full_dvf_local(
            predicted_coeff
        )

        moved = experiment_transformer(
            moving,
            predicted_flow,
        )

        image_mse = F.mse_loss(moved, fixed)
        coeff_mse = coefficient_loss(
            predicted_coeff,
            target_coeff,
        )

        total_loss = (
            IMAGE_WEIGHT * image_mse
            + DVF_WEIGHT * coeff_mse
        )

        if EXPERIMENT == "jacobian":
            jacobian_loss = sampled_jacobian_loss(
                predicted_flow
            )
            total_loss = (
                total_loss
                + JACOBIAN_WEIGHT * jacobian_loss
            )
        else:
            jacobian_loss = None

        if not torch.isfinite(total_loss):
            raise FloatingPointError(
                f"epoch {epoch}: 非有限のloss"
            )

        total_loss.backward()
        optimizer.step()

        if epoch == start_epoch or epoch % LOG_EVERY == 0:
            if DEVICE.type == "cuda":
                torch.cuda.synchronize()
                max_vram_gb = (
                    torch.cuda.max_memory_allocated()
                    / 1024**3
                )
            else:
                max_vram_gb = 0.0

            elapsed = time.perf_counter() - started_at

            extra = (
                f"motion={motion_kind}"
                if EXPERIMENT == "small_motion"
                else f"jac={jacobian_loss.item():.6f}"
            )

            print(
                f"epoch {epoch:05d}/{TOTAL_EPOCHS} "
                f"shift={shift:02d} "
                f"total={total_loss.item():.6f} "
                f"image={image_mse.item():.6f} "
                f"coeff={coeff_mse.item():.6f} "
                f"{extra} "
                f"maxVRAM={max_vram_gb:.2f}GB "
                f"elapsed={elapsed:.1f}s"
            )

        if (
            epoch % CHECKPOINT_EVERY == 0
            or epoch == TOTAL_EPOCHS
        ):
            path = save_checkpoint(
                epoch, model, optimizer
            )
            print("checkpoint保存:", path)

        del (
            moving, teacher_dvf, fixed,
            target_coeff, moving_bands, fixed_bands,
            predicted_coeff, predicted_flow, moved,
            image_mse, coeff_mse, total_loss,
        )
        if jacobian_loss is not None:
            del jacobian_loss

    print(
        "完了:",
        EXPERIMENT_ID,
        CHECKPOINT_DIR / "checkpoint_epoch_40000.pth",
    )
